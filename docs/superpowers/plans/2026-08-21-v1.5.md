# v1.5 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship v1.5 with four features: Material You dynamic theming, LAN server discovery, improved ongoing notification, and new color palettes.

**Architecture:** The app already has a `ThemeMode` + `paletteId` setting flowing through `SettingsStore` → `MonitorViewModel` → `PcHWMonitorTheme`. v1.5 adds: (1) a `dynamic` palette id resolved to `dynamicDarkColorScheme`/`dynamicLightColorScheme` on API 31+, falling back to Default below that; (2) a UDP discovery beacon on the server (same port 8765, UDP) answered by an Android `LanDiscovery` client shown in Settings as a "Scan" button; (3) an ongoing notification channel updated from the ViewModel while connected; (4) two new palettes (Candy, Nord) plus the dynamic option in settings.

**Tech Stack:** Kotlin, Jetpack Compose Material 3, DataStore, OkHttp WebSocket, Room (Android); FastAPI + uvicorn (server). Tests: JUnit 4 (Android unit tests), pytest (server).

## Global Constraints

- minSdk 24, targetSdk/compileSdk 37; versionCode 5 → 6, versionName "1.4" → "1.5" (Task 8).
- All new user-facing strings must be added to `values/strings.xml` (en) with matching keys in every `values-*` locale file (15 languages total). Existing pattern: each locale has its own `strings.xml`.
- Palette ids are lowercase strings; `PaletteDefinitions.ids` is the source of truth.
- Server discovery must not require new dependencies (stdlib `socket` only).
- UDP discovery uses the same port as TCP (8765) — no new firewall port.
- Keep `ws://` URLs unchanged; `MonitorController.isPrivateIp` guard stays.
- App must remain compatible with older servers (no discovery response = scan shows nothing).

package com.Obscrum.pchwmonitor.ui.theme

import android.os.Build
import androidx.compose.foundation.isSystemInDarkTheme
import androidx.compose.material3.MaterialTheme
import androidx.compose.material3.dynamicDarkColorScheme
import androidx.compose.material3.dynamicLightColorScheme
import androidx.compose.runtime.Composable
import androidx.compose.ui.platform.LocalContext
import com.Obscrum.pchwmonitor.data.ThemeMode

@Composable
fun PcHWMonitorTheme(
    themeMode: ThemeMode = ThemeMode.SYSTEM,
    paletteId: String = "default",
    content: @Composable () -> Unit
) {
    val darkTheme = when (themeMode) {
        ThemeMode.SYSTEM -> isSystemInDarkTheme()
        ThemeMode.LIGHT -> false
        ThemeMode.DARK -> true
    }
    val context = LocalContext.current
    val colorScheme = when {
        paletteId == "dynamic" && Build.VERSION.SDK_INT >= 31 ->
            if (darkTheme) dynamicDarkColorScheme(context) else dynamicLightColorScheme(context)
        else -> PaletteDefinitions.schemeFor(paletteId, darkTheme)
    }

    MaterialTheme(
        colorScheme = colorScheme,
        typography = Typography,
        content = content
    )
}

- [ ] **Step 3: Implement discovery.py**

```python
import json
import logging
import socket
import threading

logger = logging.getLogger("pchw.discovery")

MAGIC = b"PCHW_DISCOVER"


class DiscoveryServer:
    """Answers UDP datagrams with server identity so Android clients can find it."""

    def __init__(self, port: int, server_name: str):
        self.port = port
        self.server_name = server_name
        self._socket: socket.socket | None = None
        self._thread: threading.Thread | None = None
        self._stop = threading.Event()

    def start(self) -> None:
        if self._thread is not None:
            return
        self._stop.clear()
        self._socket = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        self._socket.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self._socket.bind(("0.0.0.0", self.port))
        self._socket.settimeout(0.5)
        self._thread = threading.Thread(target=self._run, name="pchw-discovery", daemon=True)
        self._thread.start()
        logger.info("discovery listening on udp 0.0.0.0:%d", self.port)

    def stop(self) -> None:
        self._stop.set()
        if self._socket is not None:
            try:
                self._socket.close()
            except OSError:
                pass
        if self._thread is not None:
            self._thread.join(timeout=1.0)
        self._thread = None
        self._socket = None

    def _run(self) -> None:
        assert self._socket is not None
        reply = json.dumps({"name": self.server_name, "port": self.port}).encode("utf-8")
        while not self._stop.is_set():
            try:
                data, addr = self._socket.recvfrom(1024)
            except socket.timeout:
                continue
            except OSError:
                break
            if data.strip() == MAGIC:
                try:
                    self._socket.sendto(reply, addr)
                except OSError:
                    pass
```

- [ ] **Step 3: Implement LanDiscovery.kt**

```kotlin
package com.Obscrum.pchwmonitor.data.network

import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import kotlinx.serialization.json.Json
import kotlinx.serialization.json.jsonObject
import kotlinx.serialization.json.jsonPrimitive
import java.net.DatagramPacket
import java.net.DatagramSocket
import java.net.InetAddress

data class DiscoveredServer(
    val name: String,
    val host: String,
    val port: Int,
)

private const val MAGIC = "PCHW_DISCOVER"

fun parseDiscoveryReply(raw: ByteArray, senderHost: String): DiscoveredServer? {
    if (!isPrivateIp(senderHost)) return null
    return try {
        val json = Json.parseToJsonElement(raw.decodeToString())
        val name = json.jsonObject["name"]?.jsonPrimitive?.content ?: return null
        val port = json.jsonObject["port"]?.jsonPrimitive?.content?.toIntOrNull() ?: return null
        DiscoveredServer(name = name, host = senderHost, port = port)
    } catch (_: Exception) {
        null
    }
}

private fun isPrivateIp(ip: String): Boolean {
    val parts = ip.split('.').mapNotNull { it.toIntOrNull() }
    if (parts.size != 4 || parts.any { it !in 0..255 }) return false
    val a = parts[0]
    val b = parts[1]
    return a == 10 || (a == 172 && b in 16..31) || (a == 192 && b == 168)
}

object LanDiscovery {

    suspend fun scan(port: Int = 8765, timeoutMs: Long = 1500): List<DiscoveredServer> =
        withContext(Dispatchers.IO) {
            val socket = DatagramSocket()
            socket.soTimeout = timeoutMs.toInt()
            val results = mutableListOf<DiscoveredServer>()
            try {
                val broadcast = InetAddress.getByName("255.255.255.255")
                val packet = DatagramPacket(MAGIC.toByteArray(), MAGIC.length, broadcast, port)
                socket.send(packet)
                val buffer = ByteArray(4096)
                val deadline = System.currentTimeMillis() + timeoutMs
                while (System.currentTimeMillis() < deadline) {
                    try {
                        val reply = DatagramPacket(buffer, buffer.size)
                        socket.receive(reply)
                        val host = reply.address?.hostAddress ?: continue
                        parseDiscoveryReply(reply.data.copyOf(reply.length), host)?.let { results.add(it) }
                    } catch (e: java.net.SocketTimeoutException) {
                        break
                    }
                }
            } finally {
                socket.close()
            }
            results.distinctBy { it.host }
        }
}
```

- [ ] **Step 3: Implement Notifications.kt**

```kotlin
package com.Obscrum.pchwmonitor

import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.content.Context
import androidx.core.app.NotificationCompat
import com.Obscrum.pchwmonitor.domain.model.SystemStatus
import java.util.Locale

object Notifications {
    const val CHANNEL_ID = "metrics"
    const val NOTIFICATION_ID = 1

    fun createChannel(manager: NotificationManager) {
        val channel = NotificationChannel(CHANNEL_ID, "Live metrics", NotificationManager.IMPORTANCE_LOW)
        manager.createNotificationChannel(channel)
    }

    fun update(context: Context, status: SystemStatus, serverName: String?) {
        val text = formatNotificationText(status)
        val title = if (serverName.isNullOrBlank()) "PC HW Monitor" else "PC · $serverName"
        val notification: Notification = NotificationCompat.Builder(context, CHANNEL_ID)
            .setSmallIcon(R.drawable.ic_launcher_foreground)
            .setContentTitle(title)
            .setContentText(text)
            .setStyle(NotificationCompat.BigTextStyle().bigText(text))
            .setOngoing(true)
            .setOnlyAlertOnce(true)
            .build()
        context.getSystemService(NotificationManager::class.java).notify(NOTIFICATION_ID, notification)
    }

    fun stop(context: Context) {
        context.getSystemService(NotificationManager::class.java).cancel(NOTIFICATION_ID)
    }

    fun formatNotificationText(status: SystemStatus): String {
        val cpu = status.cpu
        val gpu = status.gpu
        val ram = status.ram
        return buildString {
            if (cpu != null) {
                val temp = cpu.tempC?.let { String.format(Locale.US, " · %.0f°C", it) } ?: ""
                append("CPU ${cpu.usagePct.toInt()}%").append(temp).append('\n')
            }
            if (gpu != null) {
                val temp = gpu.tempC?.let { String.format(Locale.US, " · %.0f°C", it) } ?: ""
                append("GPU ${gpu.usagePct.toInt()}%").append(temp).append('\n')
            }
            if (ram != null) {
                append(String.format(Locale.US, "RAM %.1f/%.0f GB", ram.usedGb, ram.totalGb)).append('\n')
            }
        }.trimEnd()
    }
}
```
